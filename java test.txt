import java.sql.*;

public class EmployeeManagement {

    // Database connection details
    private static final String URL = "jdbc:mysql://localhost:3306/TEST";
    private static final String USERNAME = "your_username"; // Replace with your MySQL username
    private static final String PASSWORD = "your_password"; // Replace with your MySQL password

    public static void main(String[] args) {
        Connection conn = null;
        PreparedStatement pstmt = null;
        ResultSet rs = null;

        try {
            // Establish the connection
            conn = DriverManager.getConnection(URL, USERNAME, PASSWORD);
            System.out.println("Connected to the TEST database.");

            // 1. Insert two employee records using PreparedStatement
            String insertSQL = "INSERT INTO employee (id, name, department, salary) VALUES (?, ?, ?, ?)";
            pstmt = conn.prepareStatement(insertSQL);

            // Insert Employee 1
            pstmt.setInt(1, 101);
            pstmt.setString(2, "E1 Name");
            pstmt.setString(3, "Engineering");
            pstmt.setInt(4, 70000);
            pstmt.executeUpdate();
            System.out.println("Inserted employee E1.");

            // Insert Employee 2
            pstmt.setInt(1, 102);
            pstmt.setString(2, "E2 Name");
            pstmt.setString(3, "Sales");
            pstmt.setInt(4, 60000);
            pstmt.executeUpdate();
            System.out.println("Inserted employee E2.");

            // 2. Update the salary of one employee using UPDATE with PreparedStatement
            String updateSQL = "UPDATE employee SET salary = ? WHERE id = ?";
            pstmt = conn.prepareStatement(updateSQL);
            pstmt.setInt(1, 75000); // New salary
            pstmt.setInt(2, 101); // Employee ID to update
            pstmt.executeUpdate();
            System.out.println("Updated salary for employee 101.");

            // 3. Retrieve and display all employees using SELECT and ResultSet
            String selectSQL = "SELECT * FROM employee";
            pstmt = conn.prepareStatement(selectSQL);
            rs = pstmt.executeQuery();
            
            System.out.println("\n--- Employee Records ---");
            while (rs.next()) {
                int id = rs.getInt("id");
                String name = rs.getString("name");
                String department = rs.getString("department");
                int salary = rs.getInt("salary");
                System.out.println("ID: " + id + ", Name: " + name + ", Department: " + department + ", Salary: " + salary);
            }
            System.out.println("------------------------");

        } catch (SQLException e) {
            e.printStackTrace();
        } finally {
            // 4. Close all connections and statements properly
            try {
                if (rs != null) rs.close();
                if (pstmt != null) pstmt.close();
                if (conn != null) conn.close();
                System.out.println("\nConnections closed.");
            } catch (SQLException e) {
                e.printStackTrace();
            }
        }
    }
}
